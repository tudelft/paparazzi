223// Each platform supplies CameraLinkTransmit; this shared header stays OS-independent.
224
225extern uint8_t catia_ck_a, catia_ck_b;
226
227#define STX 0x99
228
229#define CatiaSizeOf(_payload) (_payload+5)
230
231#define CatiaPutUint8( _byte) { \
232 catia_ck_a += _byte; \
233 catia_ck_b += catia_ck_a; \
234 CameraLinkTransmit(_byte); \
235 }
236
237#define CatiaHeader(msg_id, payload_len) { \
238 CameraLinkTransmit(STX); \
239 uint8_t msg_len = CatiaSizeOf( payload_len); \
240 CameraLinkTransmit(msg_len); \
241 catia_ck_a = msg_len; catia_ck_b = msg_len; \
242 CatiaPutUint8(msg_id); \
243 }
244
245#define CatiaTrailer() { \
246 CameraLinkTransmit(catia_ck_a); \
247 CameraLinkTransmit(catia_ck_b); \
248 }
249
250#define CatiaPut1ByteByAddr( _byte) { \
251 uint8_t _x = *(_byte); \
252 CatiaPutUint8( _x); \
253 }
