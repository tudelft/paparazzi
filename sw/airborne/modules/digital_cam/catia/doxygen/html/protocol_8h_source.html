219// Each platform supplies CameraLinkTransmit; this shared header stays OS-independent.
220
221extern uint8_t catia_ck_a, catia_ck_b;
222
223#define STX 0x99
224
225#define CatiaSizeOf(_payload) (_payload+5)
226
227#define CatiaPutUint8( _byte) { \
228 catia_ck_a += _byte; \
229 catia_ck_b += catia_ck_a; \
230 CameraLinkTransmit(_byte); \
231 }
232
233#define CatiaHeader(msg_id, payload_len) { \
234 CameraLinkTransmit(STX); \
235 uint8_t msg_len = CatiaSizeOf( payload_len); \
236 CameraLinkTransmit(msg_len); \
237 catia_ck_a = msg_len; catia_ck_b = msg_len; \
238 CatiaPutUint8(msg_id); \
239 }
240
241#define CatiaTrailer() { \
242 CameraLinkTransmit(catia_ck_a); \
243 CameraLinkTransmit(catia_ck_b); \
244 }
245
246#define CatiaPut1ByteByAddr( _byte) { \
247 uint8_t _x = *(_byte); \
248 CatiaPutUint8( _x); \
249 }
